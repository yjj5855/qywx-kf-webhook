openapi: 3.0.1
info:
  title: ''

    post:
      summary: 创建外部群
      deprecated: false
      description: |-
        **功能介绍：**
        - 由机器人创建一个指定名称的外部群，同时支持拉人、修改群公告、修改群备注、使用群模板等操作

        **注意：**
        1. 该指令不会创建重名的群
      tags:

          in: query
          description: 客户端链接唯一标识
          required: true
          example: '{{robot_id}}'
          schema:
            type: string
        - name: Content-Type
          in: header
          description: ''
          required: true
          example: application/json
          schema:
            type: string
      requestBody:
        content:
          application/json:
            schema:
              type: object
              properties:
                socketType:
                  type: integer
                list:
                  type: array
                  items:
                    type: object
                    properties:
                      type:
                        type: integer
                        title: ''
                        description: 固定值=206
                      groupName:
                        type: string
                        description: 要创建的群名
                      selectList:
                        type: array
                        items:
                          type: string
                        description: 要拉入群的成员昵称
                      groupAnnouncement:
                        type: string
                        description: 群的群公告(选填)
                      groupRemark:
                        type: string
                        description: 修改群备注(选填)
                      groupTemplate:
                        type: string
                        description: 修改群模板(选填)
                    x-apifox-orders:
                      - type
                      - groupName
                      - selectList
                      - groupTemplate
                      - groupAnnouncement
                      - groupRemark
                    required:
                      - type
                      - groupName
                      - selectList
              required:
                - socketType
                - list
              x-apifox-orders:
                - socketType
                - list
            example:
              socketType: 2
              list:
                - type: 206
                  groupName: 测试群01
                  selectList:
                    - 仑哥
                  groupAnnouncement: 这里填写群的公告。WorkTool欢迎大家~
      responses:
        '200':
          description: ''
          content:
            application/json:
              schema:
                type: object
                properties:
                  code:
                    type: number
                  message:
                    type: string
                  data:
                    type: string
                required:
                  - code
                  - message
                  - data
                x-apifox-orders:
                  - code
                  - message
                  - data
          headers: {}
          x-apifox-name: 成功
      security: []
